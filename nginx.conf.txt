worker_processes 1;

events {
    worker_connections 1024;   
}

http {
    include mime.types;

    upstream goodreads_backend {
        least_conn;
        server 127.0.0.1:3000;
        server 127.0.0.1:5050;
    }


    server{
        listen 443 ssl;
        server_name localhost;

        ssl_certificate /Users/adityaparmar/My/nginx-certs/nginx-selfsigned.crt;
        ssl_certificate_key /Users/adityaparmar/My/nginx-certs/nginx-selfsigned.key;

        location / {
            proxy_pass http://goodreads_backend;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }
    }

    server {
        listen 80;
        server_name localhost;
        location / {
            return 301 https://$host$request_uri;
        }
    }
}
